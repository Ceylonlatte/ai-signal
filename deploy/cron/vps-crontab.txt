# Host crontab for the VPS (`crontab -e`). This is the form that actually runs in
# production: `docker compose exec` needs to read .env, which is root:600, so the
# cron user can't use it — `docker exec` against the container name works because
# the container already has the env baked in from compose.
PATH=/usr/local/bin:/usr/bin:/bin
# RSS daily, HN every 4 hours
0 0 * * *   docker exec ai-signal-worker-1 npm run collect:rss >> /home/deploy/aisignal-logs/rss.log 2>&1
0 */4 * * * docker exec ai-signal-worker-1 npm run collect:hn  >> /home/deploy/aisignal-logs/hn.log 2>&1
# Daily retention sweep at 04:00 (favorites preserved)
0 4 * * *   docker exec ai-signal-worker-1 npm run cleanup     >> /home/deploy/aisignal-logs/cleanup.log 2>&1
